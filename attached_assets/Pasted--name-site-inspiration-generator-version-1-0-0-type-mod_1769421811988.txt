import express from "express";
import fetch from "node-fetch";
import cheerio from "cheerio";
import path from "path";
import { fileURLToPath } from "url";

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const app = express();
app.use(express.json());
app.use(express.static("public"));

app.post("/analyze", async (req, res) => {
  const { url } = req.body;

  try {
    const response = await fetch(url);
    const html = await response.text();
    const $ = cheerio.load(html);

    const analysis = {
      title: $("title").text() || "Generated App",
      hasNav: $("nav").length > 0,
      forms: $("form").length,
      buttons: $("button").length,
      inputs: $("input").length,
      sections: $("section").length,
      headings: $("h1, h2, h3").map((i, el) => $(el).text()).get()
    };

    res.json(analysis);
  } catch (err) {
    res.status(500).json({ error: "Failed to analyze site" });
  }
});

app.listen(3000, () => {
  console.log("Running on http://localhost:3000");
});

<script>
async function generate() {
  const url = document.getElementById("url").value;
  const res = await fetch("/analyze", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ url })
  });

  const data = await res.json();

  document.getElementById("output").innerHTML = `
    <div class="card">
      <h2>${data.title}</h2>
      ${data.forms > 0 ? "<p>🔐 Secure access module generated</p>" : ""}
      ${data.hasNav ? "<p>🧭 Navigation layout created</p>" : ""}
      <p>🧩 Buttons: ${data.buttons}</p>
      <p>📄 Sections: ${data.sections}</p>
      <button>Launch New App</button>
    </div>
  `;
}
